func _ready() -> void:
    font = ThemeDB.fallback_font
    randomize()
    audio_player = AudioStreamPlayer.new()
    var generator: AudioStreamGenerator = AudioStreamGenerator.new()
    generator.mix_rate = 22050.0
    generator.buffer_length = 0.25
    audio_player.stream = generator
    audio_player.volume_db = -22.0
    add_child(audio_player)
    audio_player.play()
    audio_playback = audio_player.get_stream_playback() as AudioStreamGeneratorPlayback
    build_level()
    set_process(true)
    set_process_input(true)

func _input(event: InputEvent) -> void:
    if not (event is InputEventKey):
        return
    var keyboard_event: InputEventKey = event as InputEventKey
    if not keyboard_event.pressed or keyboard_event.echo:
        return
    var key: int = keyboard_event.keycode
    if key == KEY_M:
        sound_enabled = not sound_enabled
    if mode == "menu":
        if key == KEY_ENTER or key == KEY_SPACE:
            new_game()
        return
    if mode == "pause":
        if key == KEY_ESCAPE or key == KEY_P or key == KEY_ENTER:
            mode = "playing"
        elif key == KEY_R:
            mode = "menu"
        return
    if mode == "end":
        if key == KEY_ENTER or key == KEY_SPACE:
            new_game()
        return
    if mode != "playing":
        return
    if key == KEY_ESCAPE or key == KEY_P:
        mode = "pause"
        return
    if phase == "parkour":
        if key == KEY_R:
            respawn_character()
            announce("REAPARECER NO REINICIA EL RELOJ")
        return
    if phase == "puzzle":
        if key == KEY_BACKSPACE and puzzle_kind == "code":
            if puzzle_entry.length() > 0:
                puzzle_entry = puzzle_entry.substr(0, puzzle_entry.length() - 1)
            return
        if key == KEY_ENTER and puzzle_kind == "code":
            submit_code()
            return
        var digit: int = key_to_number(key)
        if digit >= 0:
            puzzle_input(digit)

func key_to_number(key: int) -> int:
    if key >= KEY_0 and key <= KEY_9:
        return key - KEY_0
    if key >= KEY_KP_0 and key <= KEY_KP_9:
        return key - KEY_KP_0
    return -1

func new_game() -> void:
    hearts = [MAX_HEARTS, MAX_HEARTS]
    points = [0, 0]
    victories = [0, 0]
    window_damage = [0, 0]
    round_success = [false, false]
    round_index = 0
    active_player = 0
    rival_parkour_limit = PARKOUR_LIMIT
    first_parkour_record = -1.0
    elapsed = 0.0
    fear = 0.0
    mode = "playing"
    build_level()
    start_turn()
    announce("RONDA 1 / 10 - JUGADOR 1")

func start_turn() -> void:
    if hearts[active_player] <= 0:
        complete_turn(false)
        return
    attempts = 0
    phase = "parkour"
    parkour_limit = PARKOUR_LIMIT if active_player == 0 else rival_parkour_limit
    parkour_remaining = parkour_limit
    puzzle_remaining = PUZZLE_LIMIT
    reset_puzzle()
    respawn_character()

func restart_round_after_loss() -> void:
    # El jugador continua en el MISMO nivel. Se reinician los dos relojes.
    if active_player == 0:
        first_parkour_record = -1.0
    phase = "parkour"
    parkour_remaining = parkour_limit
    puzzle_remaining = PUZZLE_LIMIT
    reset_puzzle()
    respawn_character()
    attempts += 1
    announce("REINTENTO DE RONDA - VIDA PERDIDA")

func respawn_character() -> void:
    var spawn: Rect2 = fixed_platforms[0]
    character_pos = Vector2(spawn.position.x + 72.0, spawn.position.y - 16.0)
    character_vel = Vector2.ZERO
    grounded = true
    jump_held = false
    particles.clear()

func announce(text: String) -> void:
    message = text
    message_left = 2.2

func damage_window(player: int, amount: int = 1) -> void:
    window_damage[player] = mini(3, window_damage[player] + amount)
    shake = 7.0
    flash = 0.60
    fear = 1.0

func lose_life(reason: String, restart_entire_round: bool, extra_damage: int = 0) -> void:
    if mode != "playing":
        return
    hearts[active_player] = maxi(0, hearts[active_player] - 1)
    damage_window(active_player, 1 + extra_damage)
    play_tone(145.0, 0.23)
    if hearts[active_player] <= 0:
        announce("SIN VIDAS - TURNO PERDIDO")
        complete_turn(false)
    elif restart_entire_round:
        restart_round_after_loss()
    else:
        respawn_character()
        announce(reason + " - EL TIEMPO CONTINUA")

func parkour_timeout() -> void:
    parkour_remaining = 0.0
    lose_life("TIEMPO DEL PARKOUR AGOTADO", true, 1)

func puzzle_timeout() -> void:
    puzzle_remaining = 0.0
    # Regla solicitada: vuelve AL PRINCIPIO DEL PARKOUR y pierde una vida.
    lose_life("PUZZLE SIN RESOLVER", true, 1)

func reach_parkour_finish() -> void:
    if phase != "parkour" or mode != "playing":
        return
    if active_player == 0:
        first_parkour_record = clampf(parkour_limit - parkour_remaining, 0.1, PARKOUR_LIMIT)
        rival_parkour_limit = first_parkour_record
    phase = "puzzle"
    puzzle_remaining = PUZZLE_LIMIT
    configure_puzzle()
    play_tone(675.0, 0.17)
    announce("PUZZLE DESBLOQUEADO - 60 SEGUNDOS")

func complete_puzzle() -> void:
    if phase != "puzzle" or mode != "playing":
        return
    victories[active_player] += 1
    round_success[active_player] = true
    points[active_player] += 500 + int(parkour_remaining * 5.0) + int(puzzle_remaining * 8.0)
    play_tone(820.0, 0.34)
    complete_turn(true)

func complete_turn(success: bool) -> void:
    if mode != "playing":
        return
    round_success[active_player] = success
    if active_player == 0:
        # Si el primer jugador no supero el parkour, el rival recibe 120 s.
        if first_parkour_record < 0.0:
            rival_parkour_limit = PARKOUR_LIMIT
        active_player = 1
        if hearts[1] <= 0:
            finish_round()
        else:
            start_turn()
            announce("JUGADOR 2 - %.2f S PARA EL PARKOUR" % parkour_limit)
        return
    finish_round()

func finish_round() -> void:
    round_index += 1
    if round_index >= MAX_ROUNDS or (hearts[0] <= 0 and hearts[1] <= 0):
        mode = "end"
        return
    # Se permite continuar a un jugador superviviente; el jugador sin vidas se salta.
    active_player = 0 if hearts[0] > 0 else 1
    first_parkour_record = -1.0
    rival_parkour_limit = PARKOUR_LIMIT
    round_success = [false, false]
    build_level()
    start_turn()
    announce("RONDA %d / %d - %s" % [round_index + 1, MAX_ROUNDS, ROUND_NAMES[round_index]])

func platforms_now() -> Array[Rect2]:
    var result: Array[Rect2] = fixed_platforms.duplicate()
    for entry in moving_platforms:
        var base: Rect2 = entry["rect"]
        var amount: float = sin(elapsed * float(entry["speed"])) * float(entry["amp"])
        var offset: Vector2 = Vector2(0.0,amount) if bool(entry["vertical"]) else Vector2(amount,0.0)
        result.append(Rect2(base.position + offset,base.size))
    return result

func saw_center(saw: Dictionary) -> Vector2:
    return Vector2(float(saw["x"]) + sin(elapsed * float(saw["speed"])) * float(saw["amp"]),float(saw["y"]))

func _process(delta: float) -> void:
    var dt: float = minf(delta, 0.04)
    elapsed += dt
    shake = maxf(0.0, shake - dt * 25.0)
    flash = maxf(0.0, flash - dt * 1.7)
    fear = maxf(0.0, fear - dt * 0.5)
    message_left = maxf(0.0, message_left - dt)
    puzzle_wrong = maxf(0.0, puzzle_wrong - dt)
    update_audio()
    if mode == "playing":
        if phase == "parkour":
            parkour_remaining = maxf(0.0, parkour_remaining - dt)
            if parkour_remaining <= 0.0:
                parkour_timeout()
            else:
                simulate_parkour(dt)
        elif phase == "puzzle":
            puzzle_remaining = maxf(0.0,puzzle_remaining - dt)
            puzzle_show_for = maxf(0.0, puzzle_show_for - dt)
            if puzzle_remaining <= 0.0:
                puzzle_timeout()
    for i in range(particles.size()-1, -1, -1):
        var part: Dictionary = particles[i]
        part["life"] = float(part["life"]) - dt
        part["pos"] = Vector2(part["pos"]) + Vector2(part["velocity"]) * dt
        if float(part["life"]) <= 0.0:
            particles.remove_at(i)
        else:
            particles[i] = part
    queue_redraw()

func simulate_parkour(dt: float) -> void:
    var axis: float = 0.0
    if Input.is_key_pressed(KEY_A) or Input.is_key_pressed(KEY_LEFT):
        axis -= 1.0
    if Input.is_key_pressed(KEY_D) or Input.is_key_pressed(KEY_RIGHT):
        axis += 1.0
    if axis != 0.0:
        facing = int(sign(axis))
    var run_speed: float = SPEED * (1.30 if Input.is_key_pressed(KEY_SHIFT) else 1.0)
    character_vel.x = axis * run_speed
    var jump_now: bool = Input.is_key_pressed(KEY_SPACE) or Input.is_key_pressed(KEY_W) or Input.is_key_pressed(KEY_UP)
    if grounded and jump_now and not jump_held:
        character_vel.y = JUMP
        grounded = false
        play_tone(440.0, 0.06)
    jump_held = jump_now
    character_vel.y += GRAVITY * dt
    var feet_before: float = character_pos.y + 16.0
    character_pos.x = clampf(character_pos.x + character_vel.x*dt, 12.0, 755.0)
    character_pos.y += character_vel.y*dt
    grounded = false
    if character_vel.y >= 0.0:
        for platform in platforms_now():
            if character_pos.x + 11.0 > platform.position.x and character_pos.x - 11.0 < platform.end.x and feet_before <= platform.position.y + 9.0 and character_pos.y + 16.0 >= platform.position.y:
                character_pos.y = platform.position.y - 16.0
                character_vel.y = 0.0
                grounded = true
                break
    if character_pos.y > 585.0:
        lose_life("CAIDA", false)
        return
    var foot: Vector2 = character_pos + Vector2(0.0,14.0)
    for spikes in spike_zones:
        if Rect2(spikes.position - Vector2(4.0,7.0),spikes.size+Vector2(8.0,8.0)).has_point(foot):
            lose_life("PINCHOS", false)
            return
    for saw in saws:
        if character_pos.distance_to(saw_center(saw)) < 24.0:
            lose_life("SIERRA", false)
            return
    # Puerta al final de la ultima plataforma: al llegar se activa el puzzle.
    var finish_platform: Rect2 = fixed_platforms[fixed_platforms.size()-1]
    if character_pos.x > 708.0 and grounded and character_pos.y + 16.0 <= finish_platform.position.y + 5.0:
        reach_parkour_finish()
        return
    if absf(character_vel.x) > 2.0 and grounded and randf() < 0.22:
        particles.append({"pos":character_pos + Vector2(-float(facing)*8.0,12.0),"velocity":Vector2(randf_range(-16.0,16.0),randf_range(-42.0,-14.0)),"life":0.37})
    anim_clock += dt * (12.0 if absf(character_vel.x)>2.0 else 2.0)

func set_sequence(title: String, sequence: Array, hidden: bool, seconds: float = 0.0) -> void:
    puzzle_title = title
    puzzle_kind = "memory" if hidden else "sequence"
    for number in sequence:
        puzzle_sequence.append(int(number))
    puzzle_show_for = seconds
    puzzle_hint = "MEMORIZA Y REPITE CON 1-4" if hidden else "PULSA LOS SELLOS EN EL ORDEN INDICADO"

func set_code(title: String, clue: String, answer: String) -> void:
    puzzle_title = title
    puzzle_kind = "code"
    puzzle_hint = clue
    puzzle_answer = answer

func set_switches(title: String, count: int, presses: Array) -> void:
    puzzle_title = title
    puzzle_kind = "switch"
    puzzle_hint = "CADA TECLA ACTIVA SU NODO Y EL VECINO DERECHO"
    puzzle_switches.resize(count)
    puzzle_target.resize(count)
    for i in range(count):
        puzzle_switches[i] = 0
        puzzle_target[i] = 0
    for raw_press in presses:
        var press: int = int(raw_press)
        puzzle_target[press] = 1 - puzzle_target[press]
        puzzle_target[(press+1)%count] = 1 - puzzle_target[(press+1)%count]

func puzzle_input(digit: int) -> void:
    if phase != "puzzle" or mode != "playing":
        return
    if puzzle_kind == "sequence" or puzzle_kind == "memory":
        if digit < 1 or digit > 4 or puzzle_show_for > 0.0:
            return
        if digit == puzzle_sequence[puzzle_step]:
            puzzle_step += 1
            play_tone(460.0 + float(puzzle_step)*60.0,0.07)
            if puzzle_step >= puzzle_sequence.size():
                complete_puzzle()
        else:
            puzzle_step = 0
            puzzle_wrong = 0.75
            play_tone(160.0,0.12)
        return
    if puzzle_kind == "code":
        if digit >= 0 and digit <= 9 and puzzle_entry.length() < 5:
            puzzle_entry += str(digit)
            play_tone(420.0,0.035)
            if puzzle_entry.length() >= puzzle_answer.length():
                submit_code()
        return
    if puzzle_kind == "switch":
        if digit < 1 or digit > puzzle_switches.size():
            return
        var index: int = digit-1
        puzzle_switches[index] = 1 - puzzle_switches[index]
        var next_index: int = (index+1)%puzzle_switches.size()
        puzzle_switches[next_index] = 1 - puzzle_switches[next_index]
        play_tone(360.0 + float(index)*75.0,0.07)
        if puzzle_switches == puzzle_target:
            complete_puzzle()

func submit_code() -> void:
    if puzzle_entry == puzzle_answer:
        complete_puzzle()
    else:
        puzzle_wrong = 0.75
        puzzle_entry = ""
        play_tone(165.0,0.14)

func play_tone(hz: float, duration: float) -> void:
    tone_hz = hz
    tone_samples_left = int(duration * 22050.0)
    tone_phase = 0.0

func update_audio() -> void:
    if audio_playback == null or not audio_playback.can_push_buffer(256):
        return
    var frames: PackedVector2Array = PackedVector2Array()
    for i in range(256):
        var volume: float = 0.0
        if sound_enabled and tone_samples_left > 0:
            volume = sin(tone_phase) * 0.16 * minf(float(tone_samples_left)/500.0, 1.0)
            tone_phase += tone_hz * TAU / 22050.0
            tone_samples_left -= 1
        frames.append(Vector2(volume,volume))
    audio_playback.push_buffer(frames)

func box_fill(x: float,y: float,w: float,h: float,color: Color) -> void:

func text_at(value: String,x: float,y: float,size: int = 21,color: Color = Color("bbccdd"),center: bool = false) -> void:
    var px: float = x

func panel_line(x: float,y: float,w: float,h: float,color: Color,weight: float = 2.0) -> void:

func draw_ellipse(center: Vector2, radius_x: float, radius_y: float, color: Color) -> void:
    var pts: PackedVector2Array = PackedVector2Array()
    for i in range(24):
        var a: float = float(i) * TAU / 24.0
        pts.append(center + Vector2(cos(a) * radius_x, sin(a) * radius_y))
    draw_colored_polygon(pts, color)

func draw_heart(x: float,y: float,alive: bool) -> void:
    var bitmap: Array[String] = ["01100110","11111111","11111111","01111110","00111100","00011000"]
    var tone: Color = Color("ed4059") if alive else Color("526179")
    for iy in range(bitmap.size()):
        for ix in range(8):
            if bitmap[iy][ix] == "1":

func draw_window(player: int) -> void:
    var ox: float = float(player) * PANEL_W
    var x: float = ox + 143.0
    var y: float = 76.0
    var w: float = 538.0
    var h: float = 270.0
    # No dibujamos un panel opaco encima del fondo; solo el marco, grietas y el avatar del jugador.
    panel_line(x, y, w, h, Color("2d3f53"), 4.0)
    draw_line(Vector2(x + 78.0, y + 9.0), Vector2(x + 55.0, y + 86.0), Color(0.74, 0.86, 0.96, 0.12), 3.0)
    draw_back_view_player(player, Vector2(ox + 380.0, 350.0))
    var damage: int = window_damage[player]
    if damage <= 0:
        return
    var centers: Array[Vector2] = [Vector2(x + 122.0, y + 84.0), Vector2(x + 356.0, y + 164.0), Vector2(x + 260.0, y + 110.0)]
    for crack_index in range(damage):
        var center: Vector2 = centers[crack_index]
        for branch in range(9):
            var angle: float = float(branch) * TAU / 9.0 + float(crack_index) * 0.31
            var radius: float = 84.0 + float((branch * 17 + crack_index * 19) % 74)
            var mid: Vector2 = center + Vector2.from_angle(angle) * (radius * 0.45)
            var tip: Vector2 = center + Vector2.from_angle(angle + 0.10) * radius
            mid.x = clampf(mid.x, x + 5.0, x + w - 5.0)
            mid.y = clampf(mid.y, y + 5.0, y + h - 5.0)
            tip.x = clampf(tip.x, x + 5.0, x + w - 5.0)
            tip.y = clampf(tip.y, y + 5.0, y + h - 5.0)
            draw_line(center, mid, Color("16222d"), 4.0)
            draw_line(mid, tip, Color("17232e"), 3.0)
            draw_line(center, mid, Color("b8cad8"), 1.4)
            draw_line(mid, tip, Color("a9bfcd"), 1.4)
    if damage >= 3:
        draw_colored_polygon(PackedVector2Array([Vector2(x + 18.0, y + 12.0), Vector2(x + 196.0, y + 48.0), Vector2(x + 154.0, y + 190.0), Vector2(x + 61.0, y + 232.0)]), Color(0.02, 0.03, 0.05, 0.92))
        draw_colored_polygon(PackedVector2Array([Vector2(x + 286.0, y + 18.0), Vector2(x + 468.0, y + 46.0), Vector2(x + 478.0, y + 214.0), Vector2(x + 334.0, y + 164.0)]), Color(0.02, 0.03, 0.05, 0.92))
        for shard in range(12):
            var sx: float = x + 16.0 + float(shard) * 44.0
            var length: float = 13.0 + float((shard * 19) % 37)
            draw_colored_polygon(PackedVector2Array([Vector2(sx, y), Vector2(sx + 26.0, y), Vector2(sx + 9.0, y + length)]), Color(0.59, 0.71, 0.81, 0.52))

func draw_game(ox: float) -> void:
    draw_stage_background(ox)
    if round_index > 0:
        for pl in fixed_platforms:
            draw_platform(pl,ox,false)
        for entry in moving_platforms:
            var base: Rect2 = entry["rect"]
            var amount: float = sin(elapsed*float(entry["speed"]))*float(entry["amp"])
            var offset: Vector2 = Vector2(0.0,amount) if bool(entry["vertical"]) else Vector2(amount,0.0)
            draw_platform(Rect2(base.position+offset,base.size),ox,true)
        for spikes in spike_zones:
            draw_spikes(ox,spikes)
        for saw in saws:
            draw_saw(ox,saw_center(saw))
    var last: Rect2 = fixed_platforms[fixed_platforms.size()-1]
    var door_x: float = ox+714.0
    var door_y: float = STAGE_TOP+last.position.y-60.0
    box_fill(door_x,door_y,34.0,60.0,Color("090f1a"))
    panel_line(door_x,door_y,34.0,60.0,STAGE_COLORS[round_index],3.0)
    box_fill(door_x+26.0,door_y+27.0,3.0,7.0,STAGE_COLORS[round_index])
    # Personaje pixel art original.
    var sprite: Texture2D = runner[int(anim_clock)%runner.size()]
    var actor_pos: Vector2 = Vector2(ox+character_pos.x-12.0,STAGE_TOP+character_pos.y-16.0)
    if facing < 0:
        draw_set_transform(actor_pos+Vector2(24.0,0.0),0.0,Vector2(-1.0,1.0))
        draw_texture(sprite,Vector2.ZERO)
        draw_set_transform(Vector2.ZERO)
    else:
        draw_texture(sprite,actor_pos)
    for particle in particles:
        var p: Vector2 = particle["pos"]
        box_fill(ox+p.x,STAGE_TOP+p.y,3.0,3.0,Color(0.82,0.86,0.92,clampf(float(particle["life"])*2.0,0.0,1.0)))
    draw_stage_hud(ox)
